import { Offcanvas, Button } from '@nexusui/core'

function App() {
  const [open, setOpen] = useState(false)

  return (
    <>
      <Button onClick={() => setOpen(true)}>
        Ouvrir le panneau
      </Button>

      <Offcanvas open={open} onClose={() => setOpen(false)}>
        Contenu du panneau

  )
}

Offcanvas

Panneau modal avec animation depuis le bas

Le composant Offcanvas affiche un panneau qui glisse depuis le bas de l'écran. Parfait pour les formulaires, les détails de produits, ou tout contenu nécessitant l'attention de l'utilisateur.

Installation

npm install @nexusui/core

Import

import { Offcanvas } from '@nexusui/core'

Utilisation

function Example() {
  const [open, setOpen] = useState(false)

  return (
    <>
      <Button onClick={() => setOpen(true)}>
        Ouvrir
      </Button>

      <Offcanvas open={open} onClose={() => setOpen(false)}>
        <Offcanvas.Header>Titre</Offcanvas.Header>
        <Offcanvas.Body>
          Contenu du panneau
        </Offcanvas.Body>
      </Offcanvas>
    </>
  )
}

Propriétés disponibles pour personnaliser le comportement du composant.

Prop Type Défaut Description
open boolean false Contrôle la visibilité
onClose () => void — Callback de fermeture
height string '85vh' Hauteur du panneau
dismissable boolean true Fermeture par backdrop
closeOnEsc boolean true Fermeture par Escape

Exemples d'utilisation dans différents scénarios.

Formulaire de contact

<Offcanvas open={showContact}>
  <Offcanvas.Header>Nous contacter</Offcanvas.Header>
  <Offcanvas.Body>
    <ContactForm />
  </Offcanvas.Body>
  <Offcanvas.Footer>
    <Button variant='primary'>Envoyer</Button>
  </Offcanvas.Footer>
</Offcanvas>

Détails produit

<Offcanvas open={showDetails} height='70vh'>
  <Offcanvas.Header>
    <ProductTitle product={product} />
  </Offcanvas.Header>
  <Offcanvas.Body>
    <ProductGallery images={product.images} />
    <ProductInfo product={product} />
  </Offcanvas.Body>
</Offcanvas>

Menu de navigation
